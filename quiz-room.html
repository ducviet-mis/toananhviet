function toggleFlag(qNum) { const box = document.getElementById(`flag-box-${qNum}`); // Chỉ đơn thuần bật / tắt ẩn hiện box chọn lý do box.classList.toggle('show'); } function toggleFlagReason(qNum, reasonType, isChecked) { const row = document.getElementById(`q-row-${qNum}`); const btn = document.getElementById(`flag-btn-${qNum}`); if (reasonType === 'tutor') { if (isChecked) tutorHelpQuestions[qNum] = true; else delete tutorHelpQuestions[qNum]; } // Kiểm tra xem có ít nhất 1 checkbox nào đang được tích không const reviewChecked = document.getElementById(`chk-review-${qNum}`)?.checked; const tutorChecked = document.getElementById(`chk-tutor-${qNum}`)?.checked; if (reviewChecked || tutorChecked) { flaggedQuestions[qNum] = true; row.classList.add('flagged-row'); // Giữ viền vàng/cam sáng btn.classList.add('active'); // Cờ vẫn sáng nổi bật } else { delete flaggedQuestions[qNum]; row.classList.remove('flagged-row'); // Tắt viền btn.classList.remove('active'); // Tắt cờ } }
